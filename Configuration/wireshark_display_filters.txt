# Custom Wireshark Display Filters used in Analysis
IP_Filter = ip.addr == 192.168.0.6
ICMP_Filter = icmp
DNS_Filter = dns
HTTP_Filter = tcp.port == 8080
TCP_Handshake = tcp.flags.syn == 1 && tcp.flags.ack == 0
Port_Reset = tcp.flags.reset == 1
Combined_Recon = tcp.flags.syn == 1 || tcp.flags.reset == 1
